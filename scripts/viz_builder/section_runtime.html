export function useChatStream() {
  const currentMessage = ref('');
  const steps = ref([]);
  const citations = ref>(new Map());
  const isStreaming = ref(false);
  let abortController: AbortController | null = null;

  const sendMessage = async (prompt: string, sessionId: string) => {
    isStreaming.value = true;
    abortController = new AbortController();

    const response = await fetch('/api/v1/chat/stream', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ prompt, session_id: sessionId }),
      signal: abortController.signal
    });

    const reader = response.body?.getReader();
    const decoder = new TextDecoder();
    while (true) {
      const { done, value } = await reader!.read();
      if (done) break;
      const chunk = decoder.decode(value, { stream: true });
      processSseChunk(chunk, (event) => {
        if (event.type === 'token') currentMessage.value += event.content;
        else if (event.type === 'step') steps.value.push(event.step);
        else if (event.type === 'citations') registerCitations(event.data);
      });
    }
    isStreaming.value = false;
  };

  const stopGeneration = () => {
    abortController?.abort();
    isStreaming.value = false;
  };

  return { currentMessage, steps, citations, isStreaming, sendMessage, stopGeneration };
}

import { defineStore } from 'pinia';
import type { Message, RagStep, GroupedRagStep, RagTrace } from '@/types/chat';

export const useChatStore = defineStore('chat', {
  state: () => ({
    messages: [] as Message[],
    messagesBySession: {} as Record,
    userInput: '',
    isLoading: false,
    sessionId: 'session_' + Date.now(),
    streamingSessionId: null as string | null,
    abortController: null as AbortController | null,
  }),

  getters: {
    isViewingStreamingSession(state): boolean {
      return state.isLoading && state.streamingSessionId === state.sessionId;
    },
    isInputLocked(state): boolean {
      return state.isLoading && state.streamingSessionId !== state.sessionId;
    }
  },

  actions: {
    async sendUserMessage(content: string) {
      if (!content.trim() || this.isLoading) return;
      const requestSessionId = this.sessionId;
      this.isLoading = true;
      this.streamingSessionId = requestSessionId;
      this.abortController = new AbortController();

      // 1. 压入用户提问与 Bot 占位消息
      const sessionMsgs = this.ensureSessionMessages(requestSessionId);
      sessionMsgs.push({ id: 'msg_' + Date.now(), text: content, isUser: true, timestamp: Date.now() });
      const botMsgIdx = sessionMsgs.push({
        id: 'msg_bot_' + Date.now(),
        text: '',
        isUser: false,
        isThinking: true,
        groupedRagSteps: [],
        ragTrace: null,
        timestamp: Date.now()
      }) - 1;

      try {
        const response = await fetch('/api/v1/chat/stream', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ prompt: content, session_id: requestSessionId }),
          signal: this.abortController.signal
        });

        const reader = response.body?.getReader();
        const decoder = new TextDecoder();
        let buffer = '';

        while (reader) {
          const { done, value } = await reader.read();
          if (done) break;
          buffer += decoder.decode(value, { stream: true });
          
          let eventEndIndex: number;
          while ((eventEndIndex = buffer.indexOf('

')) >= 0) {
            const eventStr = buffer.slice(0, eventEndIndex);
            buffer = buffer.slice(eventEndIndex + 2);

            if (eventStr.startsWith('data: ')) {
              const dataStr = eventStr.slice(6);
              if (dataStr === '[DONE]') break;
              
              const data = JSON.parse(dataStr);
              const botMsg = sessionMsgs[botMsgIdx];

              if (data.type === 'content') {
                botMsg.isThinking = false;
                botMsg.text += data.content; // 增量打字机渲染
              } else if (data.type === 'rag_step') {
                botMsg.groupedRagSteps = this.appendRagStepToGroups(botMsg.groupedRagSteps, data.step);
              } else if (data.type === 'rag_trace') {
                botMsg.ragTrace = data.trace; // 挂载检索指标与金标对账
              }
            }
          }
        }
      } catch (err: any) {
        const botMsg = sessionMsgs[botMsgIdx];
        botMsg.isThinking = false;
        if (err.name === 'AbortError') {
          botMsg.text += '\n\n*(生成已被用户手动终止)*';
        } else {
          botMsg.text += `\n[通信错误: ${err.message}]`;
        }
      } finally {
        this.isLoading = false;
        this.streamingSessionId = null;
        this.abortController = null;
      }
    }
  }
});

为什么必须关闭代理缓冲（proxy_buffering off）？
默认情况下，Nginx 会对上游 HTTP 响应开启缓冲区（一般为 4KB~8KB）。由于 SSE 的每个 Token 只有几十字节，Nginx 会将 Token 扣押在内存中直到填满缓冲区才一次性冲刷，导致前端呈现为严重的卡顿和“瞬间吐出一大段”现象。如下配置彻底解决了中间件拦截问题，保证全链路 60 FPS 顺畅输出。
Nginx / /etc/nginx/conf.d/supermew.conf - 生产级 SSE 流式反向代理配置 
server {
    listen 80;
    server_name rag.supermew.enterprise.internal;

    # 静态资源与 Vue 3 前端 SPA
    location / {
        root /usr/share/nginx/html;
        try_files $uri $uri/ /index.html;
    }

    # 核心 SSE 流式问答接口（必须禁用代理缓冲与缓存）
    location /api/v1/chat/stream {
        proxy_pass http://supermew-backend:8000;
        proxy_http_version 1.1;

        # 核心关键参数：禁用缓冲区，确保上游产生每个 Token 立即推送到客户端
        proxy_buffering off;
        proxy_cache off;
        chunked_transfer_encoding on;

        # 保持连接并防止超长思考导致连接中断（允许单次复杂检索思考长达 10 分钟）
        proxy_read_timeout 600s;
        proxy_send_timeout 600s;
        proxy_set_header Connection '';

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

2. 全生命周期 SSE 消费：支持 type: content（打字机文字流）、type: rag_step（折叠思维链）、type: rag_trace（检索元数据）、type: session（会话标题自动同步）与 [DONE] 严格结束标记。
3. 客户端主动中断：基于原生 AbortController 实现随时中止生成，中断后自动保持局部历史完好并同步到后端状态账本。

import { defineStore } from 'pinia';
import type { Message, RagStep, GroupedRagStep, RagTrace } from '@/types/chat';

export const useChatStore = defineStore('chat', {
  state: () => ({
    messages: [] as Message[],
    messagesBySession: {} as Record,
    userInput: '',
    isLoading: false,
    sessionId: 'session_' + Date.now(),
    streamingSessionId: null as string | null,
    abortController: null as AbortController | null,
  }),

  getters: {
    isViewingStreamingSession(state): boolean {
      return state.isLoading && state.streamingSessionId === state.sessionId;
    },
    isInputLocked(state): boolean {
      return state.isLoading && state.streamingSessionId !== state.sessionId;
    }
  },

  actions: {
    async sendUserMessage(content: string) {
      if (!content.trim() || this.isLoading) return;
      const requestSessionId = this.sessionId;
      this.isLoading = true;
      this.streamingSessionId = requestSessionId;
      this.abortController = new AbortController();

      // 1. 压入用户提问与 Bot 占位消息
      const sessionMsgs = this.ensureSessionMessages(requestSessionId);
      sessionMsgs.push({ id: 'msg_' + Date.now(), text: content, isUser: true, timestamp: Date.now() });
      const botMsgIdx = sessionMsgs.push({
        id: 'msg_bot_' + Date.now(),
        text: '',
        isUser: false,
        isThinking: true,
        groupedRagSteps: [],
        ragTrace: null,
        timestamp: Date.now()
      }) - 1;

      try {
        const response = await fetch('/api/v1/chat/stream', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ prompt: content, session_id: requestSessionId }),
          signal: this.abortController.signal
        });

        const reader = response.body?.getReader();
        const decoder = new TextDecoder();
        let buffer = '';

        while (reader) {
          const { done, value } = await reader.read();
          if (done) break;
          buffer += decoder.decode(value, { stream: true });
          
          let eventEndIndex: number;
          while ((eventEndIndex = buffer.indexOf('

')) >= 0) {
            const eventStr = buffer.slice(0, eventEndIndex);
            buffer = buffer.slice(eventEndIndex + 2);

            if (eventStr.startsWith('data: ')) {
              const dataStr = eventStr.slice(6);
              if (dataStr === '[DONE]') break;
              
              const data = JSON.parse(dataStr);
              const botMsg = sessionMsgs[botMsgIdx];

              if (data.type === 'content') {
                botMsg.isThinking = false;
                botMsg.text += data.content; // 增量打字机渲染
              } else if (data.type === 'rag_step') {
                botMsg.groupedRagSteps = this.appendRagStepToGroups(botMsg.groupedRagSteps, data.step);
              } else if (data.type === 'rag_trace') {
                botMsg.ragTrace = data.trace; // 挂载检索指标与金标对账
              }
            }
          }
        }
      } catch (err: any) {
        const botMsg = sessionMsgs[botMsgIdx];
        botMsg.isThinking = false;
        if (err.name === 'AbortError') {
          botMsg.text += '\n\n*(生成已被用户手动终止)*';
        } else {
          botMsg.text += `\n[通信错误: ${err.message}]`;
        }
      } finally {
        this.isLoading = false;
        this.streamingSessionId = null;
        this.abortController = null;
      }
    }
  }
});

为什么必须关闭代理缓冲（proxy_buffering off）？
默认情况下，Nginx 会对上游 HTTP 响应开启缓冲区（一般为 4KB~8KB）。由于 SSE 的每个 Token 只有几十字节，Nginx 会将 Token 扣押在内存中直到填满缓冲区才一次性冲刷，导致前端呈现为严重的卡顿和“瞬间吐出一大段”现象。如下配置彻底解决了中间件拦截问题，保证全链路 60 FPS 顺畅输出。
Nginx / /etc/nginx/conf.d/supermew.conf - 生产级 SSE 流式反向代理配置 
server {
    listen 80;
    server_name rag.supermew.enterprise.internal;

    # 静态资源与 Vue 3 前端 SPA
    location / {
        root /usr/share/nginx/html;
        try_files $uri $uri/ /index.html;
    }

    # 核心 SSE 流式问答接口（必须禁用代理缓冲与缓存）
    location /api/v1/chat/stream {
        proxy_pass http://supermew-backend:8000;
        proxy_http_version 1.1;

        # 核心关键参数：禁用缓冲区，确保上游产生每个 Token 立即推送到客户端
        proxy_buffering off;
        proxy_cache off;
        chunked_transfer_encoding on;

        # 保持连接并防止超长思考导致连接中断（允许单次复杂检索思考长达 10 分钟）

        proxy_send_timeout 600s;
        proxy_set_header Connection '';

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

🌐 生产级 Nginx 反向代理配置规范与心跳防断连机制 (Nginx Production Directives) DevOps & Infra
为什么很多团队的 SSE 在生产上线后“完全不流”？
在生产架构中，SSE 必须穿透 Nginx 反向代理与云厂商负载均衡器。如果未显式禁用缓冲（proxy_buffering off），Nginx 默认会积攒满 4KB 或等到请求完全结束才一次性刷新，打字机即时感彻底丧失；同时，超过 60 秒无数据传输时防火墙会静默掐断连接。SuperMew 规范了标准的生产代理指令集与 15s 保活心跳： 
# /etc/nginx/conf.d/supermew_sse.conf
location /api/v1/chat/stream {
    proxy_pass http://supermew_fastapi_cluster;
    proxy_http_version 1.1;
    proxy_set_header Connection "";               # 启用 HTTP/1.1 长连接连接池
    proxy_set_header Host System.Management.Automation.Internal.Host.InternalHost;
    proxy_set_header X-Real-IP ;
    proxy_set_header X-Forwarded-For ;

    # 核心流式指令：彻底关闭代理缓冲与缓存
    proxy_buffering off;                          # 严禁积攒响应包，有字节立即推送
    proxy_cache off;                              # 禁用静态代理缓存
    proxy_read_timeout 600s;                      # 允许长思考或多跳 Agent 耗时，防止 60s 报错
    proxy_send_timeout 600s;
    chunked_transfer_encoding on;                 # 启用分块传输编码
    proxy_set_header X-Accel-Buffering "no";      # 显式告知上游反向代理严禁二次缓冲
}

📜 PII 敏感信息脱敏正则引擎源码 (backend/chat/runtime.py)
class PIISanitizer:
    """中文场景企业级 PII 脱敏器：支持身份证、手机号、银行卡号动态掩码。"""
    _ID_CARD_RE = re.compile(r"(?<!\d)([1-9]\d{5}(?:18|19|20)\d{2}(?:0[1-9]|1[0-2])(?:0[1-9]|[12]\d|3[01])\d{3}[\dXx])(?!\d)")
    _MOBILE_RE = re.compile(r"(?<!\d)(1[3-9]\d{9})(?!\d)")
    _BANK_CARD_RE = re.compile(r"(?<!\d)([1-9]\d{15,18})(?!\d)")

    @classmethod
    def mask_text(cls, text: str) -> str:
        """将敏感个人信息替换为合规掩码格式，保留首尾标识供业务对照。"""
        if not text:
            return ""
        # 身份证掩码: 110101********1234
        text = cls._ID_CARD_RE.sub(lambda m: m.group(1)[:6] + "********" + m.group(1)[-4:], text)
        # 手机号掩码: 138****5678
        text = cls._MOBILE_RE.sub(lambda m: m.group(1)[:3] + "****" + m.group(1)[-4:], text)
        # 银行卡掩码: 6222***********1234
        text = cls._BANK_CARD_RE.sub(lambda m: m.group(1)[:4] + "***********" + m.group(1)[-4:], text)
        return text
