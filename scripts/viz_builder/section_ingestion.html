class MineruClient:
    """MinerU Gradio 异步解析客户端：具备指数退避重试、网络抖动恢复与本地缓存保护机制。"""
    def __init__(self, base_url: str = "http://10.0.4.15:7860", timeout_s: float = 600.0):
        self.client = Client(base_url)
        self.timeout_s = timeout_s

    def parse_pdf_async(self, file_path: Path, output_dir: Path) -> Path:
        """提交解析任务并采用自适应退避轮询，直到任务完成或硬超时熔断。"""
        job = self.client.submit(
            handle_file(str(file_path)),
            api_name="/predict_pdf"
        )
        start_time = time.time()
        backoff = 1.0  # 初始退避 1 秒

        while not job.done():
            elapsed = time.time() - start_time
            if elapsed > self.timeout_s:
                job.cancel()
                raise TimeoutError(f"MinerU parse timeout after {elapsed:.1f}s for {file_path.name}")
            
            time.sleep(backoff)
            backoff = min(backoff * 1.5, 10.0)  # 指数退避，上限 10 秒，保护网络与 Gradio 队列

        result_zip_path = job.result()
        # 原子解压并校验 Markdown 产物与图片资源
        return self._extract_and_validate(Path(result_zip_path), output_dir)

# 正则匹配中英文句子结尾标点符号，防止在逗号、数字小数点或单词中间发生粗暴截断
_SENTENCE_END_RE = re.compile(r"(?<=[.!?。！？；])(?:["'\)\]]+)?(?=\s|$)")

class SentenceSpan(BaseModel):
    start: int
    end: int
    text: str

def split_sentences_with_spans(text: str) -> List[SentenceSpan]:
    # 将超长正文按自然句子边界切分为带偏移量的句子跨度序列
    spans: List[SentenceSpan] = []
    start = 0
    for match in _SENTENCE_END_RE.finditer(text):
        end = match.end()
        sentence = text[start:end].strip()
        if sentence:
            spans.append(SentenceSpan(start=start, end=end, text=sentence))
        start = end
    if start < len(text):
        tail = text[start:].strip()
        if tail:
            spans.append(SentenceSpan(start=start, end=len(text), text=tail))
    return spans

📐 滑动窗口余弦相似度断点算法推导与数据库存储方案 (Semantic Chunking & Storage) Math & Storage
语义向量断点算法 (Semantic Window Breakpoint)
在段落长度超过 L3 上限（800 字符）且存在多意群时，SuperMew 启动滑动窗口语义切分。以连续 W=3 个句子构成左窗口与右窗口，计算两者的余弦相似度距离。当距离出现局部峰值（即语义剧烈转折）且超过整篇段落的 20th 百分位数时，标记为最佳切分断点。 
def find_semantic_breakpoints(sentences: List[SentenceSpan], embeddings: List[List[float]], 
                              window_size: int = 3, percentile: float = 20.0) -> List[int]:
    """基于双滑动窗口内积距离计算语义突变断点。"""
    distances = []
    n = len(sentences)
    if n <= window_size * 2:
        return []  # 句子数过少，直接走标点保底切分

    for i in range(window_size, n - window_size):
        # 聚合左窗口与右窗口的平均语义向量
        left_vec = np.mean(embeddings[i - window_size : i], axis=0)
        right_vec = np.mean(embeddings[i : i + window_size], axis=0)
        # 计算余弦距离: 1 - cosine_similarity
        cos_sim = np.dot(left_vec, right_vec) / (np.linalg.norm(left_vec) * np.linalg.norm(right_vec) + 1e-8)
        distances.append(1.0 - cos_sim)

    # 动态阈值：选取局部峰值大于 20th 百分位数的断点
    threshold = np.percentile(distances, 100 - percentile)
    breakpoints = [window_size + idx for idx, d in enumerate(distances) if d >= threshold]
    return breakpoints

-- PostgreSQL 16: parent_chunks 表定义与三级树状索引
CREATE TABLE parent_chunks (
    chunk_id VARCHAR(64) PRIMARY KEY,          -- 块唯一指纹: doc_{doc_id}_l{level}_{idx}
    doc_id VARCHAR(64) NOT NULL,               -- 所属文档全局唯一 ID
    chunk_level SMALLINT NOT NULL,             -- 块级别: 1 (L1章节 2400), 2 (L2块 1600), 3 (L3叶子 800)
    parent_chunk_id VARCHAR(64),               -- 父级块指纹 (L3 指向 L2, L2 指向 L1, L1 为 NULL)
    header_path TEXT NOT NULL,                 -- 标题层级路径 (例: '# 第一章 > ## 1.2 报销标准')
    content TEXT NOT NULL,                     -- 完整 Markdown 文本内容
    token_count INT NOT NULL,                  -- Token 真实消耗统计
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_parent FOREIGN KEY (parent_chunk_id) REFERENCES parent_chunks(chunk_id) ON DELETE CASCADE
);

-- 高性能回查复合索引: 支撑检索命中 L3 后的秒级父级折叠提升 (P99 < 1.5ms)
CREATE INDEX idx_parent_chunks_lookup ON parent_chunks (doc_id, chunk_level, parent_chunk_id);
CREATE INDEX idx_parent_chunks_header ON parent_chunks USING gin (to_tsvector('simple', header_path));
📑 DocumentLoader 结构化 Markdown AST 解析与三级块切分 (backend/indexing/document_loader.py) Markdown Loader
Markdown 标题树递归遍历与语义保护
DocumentLoader 基于 Markdown 语法解析器递归提取 H1-H6 标题树，将文档按语义章节拆解。对标题下的段落文本计算字符容量，严格按照 L1 (2400) / L2 (1600) / L3 (800) 规则分层切分，并在切分过程中注入全路径面包屑标题（如 # 财务制度 > ## 报销标准 > ### 餐饮限额）。 
class DocumentLoader:
    """工业级 Markdown 结构化切分加载器：维护标题层级栈，避免表格与标题被硬截断。"""
    def __init__(self, config: SemanticChunkingConfig = None):
        self.config = config or SemanticChunkingConfig()

    def load_and_split(self, markdown_text: str, doc_id: str) -> list[Document]:
        """解析 Markdown 文本并生成具备父子指针的三级块列表。"""
        sections = self._parse_markdown_headings(markdown_text)
        all_chunks = []

        for sec_idx, section in enumerate(sections):
            header_path = section["header_path"]
            content = section["content"]
            
            # 1. 生成 L1 章节级宏观父块 (最大 2400 字符，重叠 400)
            l1_chunks = self._chunk_level(content, size=2400, overlap=400, level=1)
            
            for l1_idx, l1_text in enumerate(l1_chunks):
                l1_id = f"doc_{doc_id}_l1_{sec_idx}_{l1_idx}"
                all_chunks.append(Document(
                    page_content=l1_text,
                    metadata={"chunk_id": l1_id, "doc_id": doc_id, "chunk_level": 1, "header_path": header_path}
                ))

                # 2. 生成 L2 语义中观父块 (最大 1600 字符，重叠 200)
                l2_chunks = self._chunk_level(l1_text, size=1600, overlap=200, level=2)
                for l2_idx, l2_text in enumerate(l2_chunks):
                    l2_id = f"doc_{doc_id}_l2_{sec_idx}_{l1_idx}_{l2_idx}"
                    all_chunks.append(Document(
                        page_content=l2_text,
                        metadata={"chunk_id": l2_id, "doc_id": doc_id, "chunk_level": 2, 
                                  "parent_chunk_id": l1_id, "header_path": header_path}
                    ))

                    # 3. 生成 L3 检索叶子块 (最大 800 字符，重叠 100，使用语义标点断句)
                    l3_chunks = split_prose_by_sentences(l2_text, max_chars=800, overlap=100)
                    for l3_idx, l3_text in enumerate(l3_chunks):
                        l3_id = f"doc_{doc_id}_l3_{sec_idx}_{l1_idx}_{l2_idx}_{l3_idx}"
                        all_chunks.append(Document(
                            page_content=l3_text,
                            metadata={"chunk_id": l3_id, "doc_id": doc_id, "chunk_level": 3, 
                                      "parent_chunk_id": l2_id, "header_path": header_path}
                        ))
                        
        return all_chunks
